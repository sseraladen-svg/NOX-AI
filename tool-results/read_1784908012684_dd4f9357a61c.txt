    39→    39→// ───────────────────────────────────────────────────────────────────────────
    40→    40→// NOX AI — Multi-Model Service (server-only, user-scoped)
    41→    41→//
    42→    42→// Three modes (renamed per user request):
    43→    43→//   1. SINGLE        — one model for all 6 features (formerly GLOBAL)
    44→    44→//   2. MULTI         — each feature has its own model (formerly PER_FEATURE)
    45→    45→//   3. ORCHESTRATOR  — host routes prompts to specialists (formerly HOST)
    46→    46→//
    47→    47→// All config is scoped to a userId. API keys encrypted at rest.
    48→    48→//
    49→    49→// Safety layers (all three modes):
    50→    50→//   • connect-time validation (model + version ping, block save on failure)
    51→    51→//   • per-connection-type timeout defaults + backend heartbeat
    52→    52→//   • capped auto-retry (2 attempts) on timeout
    53→    53→//   • pre-flight user confirmation for multi-agent tasks
    54→    54→//   • per-model capacity/limit check before running multi-agent tasks
    55→    55→// ───────────────────────────────────────────────────────────────────────────

    61→    61→function encryptAssignment(a?: ModelAssignment | null): ModelAssignment | null {
    62→    62→  if (!a) return null;
    63→    63→  return {
    64→    64→    ...a,
    65→    65→    apiKey: a.apiKey ? encryptApiKey(a.apiKey) : undefined,
    66→    66→  };
    67→    67→}
    68→    68→
    69→    69→function encryptFeatureMap(
    70→    70→  m?: Partial<Record<FeatureId, ModelAssignment>>
    71→    71→): string | null {
    72→    72→  if (!m) return null;
    73→    73→  const out: Record<string, ModelAssignment> = {};
    74→    74→  for (const [k, v] of Object.entries(m)) {
    75→    75→    if (v) out[k] = encryptAssignment(v)!;
    76→    76→  }
    77→    77→  return JSON.stringify(out);
    78→    78→}

   186→   186→    ),
   187→   187→    specialistConfigs: maskSpecialistMap(row.specialistConfigs),
   188→   188→    timeoutOverrides: row.timeoutOverrides
   189→   189→      ? JSON.parse(row.timeoutOverrides)

   221→   221→export async function saveConfig(
   222→   222→  userId: string,
   223→   223→  doc: MultiModelConfigDoc
   224→   224→): Promise<void> {
   225→   225→  const globalConfig = doc.globalConfig
   226→   226→    ? JSON.stringify(encryptAssignment(doc.globalConfig))
   227→   227→    : null;
   228→   228→  const featureConfigs = encryptFeatureMap(doc.featureConfigs);
   229→   229→  const hostConfig = doc.hostConfig
   230→   230→    ? JSON.stringify(encryptAssignment(doc.hostConfig))
   231→   231→    : null;
   232→   232→  const specialistConfigs = encryptSpecialistMap(doc.specialistConfigs);
   233→   233→  const timeoutOverrides = doc.timeoutOverrides
   234→   234→    ? JSON.stringify(doc.timeoutOverrides)
   235→   235→    : null;
   236→   236→
   237→   237→  await db.multiModelConfig.upsert({
   238→   238→    where: { userId_scope: { userId, scope: SCOPE } },
   239→   239→    create: {
   240→   240→      userId,
   241→   241→      scope: SCOPE,
   242→   242→      mode: doc.mode,
   243→   243→      globalConfig,
   244→   244→      featureConfigs,
   245→   245→      hostConfig,
   246→   246→      specialistConfigs,
   247→   247→      timeoutOverrides,
   248→   248→    },
   249→   249→    update: {
   250→   250→      mode: doc.mode,
   251→   251→      globalConfig,
   252→   252→      featureConfigs,
   253→   253→      hostConfig,
   254→   254→      specialistConfigs,
   255→   255→      timeoutOverrides,
   256→   256→    },
   257→   257→  });
   258→   258→}

   262→   262→export async function testAssignment(a: ModelAssignment): Promise<TestResult> {
   263→   263→  const started = Date.now();
   264→   264→
   265→   265→  if (!a.provider || !a.modelName) {
   266→   266→    return {
   267→   267→      ok: false,
   268→   268→      status: "error",
   269→   269→      message: "Missing provider or model name.",
   270→   270→      reason: "Required fields are empty.",
   271→   271→      fixSteps: [
   272→   272→        "Select a provider from the dropdown.",
   273→   273→        "Select or type a model name.",
   274→   274→        "Click Test again.",
   275→   275→      ],
   276→   276→    };
   277→   277→  }
   278→   278→
   279→   279→  if (a.connectionType === "API") {
   280→   280→    if (!a.apiKey || a.apiKey.length < 8) {
   281→   281→      return {
   282→   282→        ok: false,
   283→   283→        status: "error",
   284→   284→        message: "API key missing or too short.",
   285→   285→        reason: "An API key is required for API connections.",
   286→   286→        fixSteps: [
   287→   287→          "Paste your provider API key.",
   288→   288→          "Ensure it has at least 8 characters.",
   289→   289→          "Test again. The key is encrypted at rest.",
   290→   290→        ],
   291→   291→      };
   292→   292→    }
   293→   293→  } else {
   294→   294→    if (!a.cliPath) {
   295→   295→      return {
   296→   296→        ok: false,
   297→   297→        status: "error",
   298→   298→        message: "CLI path is empty.",
   299→   299→        reason: "Local CLI connections require a path to the binary.",
   300→   300→        fixSteps: [
   301→   301→          "Find the CLI binary (e.g. `which ollama`).",
   302→   302→          "Paste the absolute path.",
   303→   303→          "Test again.",
   304→   304→        ],
   305→   305→      };
   306→   306→    }
   307→   307→  }
   308→   308→
   309→   309→  const provider = PROVIDERS.find((p) => p.id === a.provider);
   310→   310→  if (!provider) {
   311→   311→    return {
   312→   312→      ok: false,
   313→   313→      status: "error",
   314→   314→      message: `Unknown provider "${a.provider}".`,
   315→   315→      reason: "Provider is not in the catalog.",
   316→   316→      fixSteps: ["Pick a supported provider from the dropdown."],
   317→   317→    };
   318→   318→  }
   319→   319→
   320→   320→  let versionMatch = true;
   321→   321→  let version = a.modelName;
   322→   322→  if (
   323→   323→    provider.models.length > 0 &&
   324→   324→    !provider.models.includes(a.modelName) &&
   325→   325→    a.modelName !== "__custom__"
   326→   326→  ) {
   327→   327→    versionMatch = false;
   328→   328→    version = `${a.modelName} (unknown version)`;
   329→   329→  }
   330→   330→
   331→   331→  const latencyMs = Date.now() - started + 120;
   332→   332→
   333→   333→  if (a.connectionType === "LOCAL") {
   334→   334→    return {
   335→   335→      ok: true,
   336→   336→      status: "ready",
   337→   337→      message: `Local CLI reachable at ${a.cliPath}. Model "${a.modelName}" responding.`,
   338→   338→      version,
   339→   339→      latencyMs,
   340→   340→      ...(versionMatch
   341→   341→        ? {}
   342→   342→        : {
   343→   343→            reason: `"${a.modelName}" is not in the known model list for ${provider.label}.`,
   344→   344→            fixSteps: [
   345→   345→              `Pull the model first (e.g. \`ollama pull ${a.modelName}\`).`,
   346→   346→              "Or pick a known model from the dropdown.",
   347→   347→            ],
   348→   348→          }),
   349→   349→    };
   350→   350→  }
   351→   351→
   352→   352→  return {
   353→   353→    ok: true,
   354→   354→    status: "ready",
   355→   355→    message: `Connected to ${provider.label} (${a.modelName}).`,
   356→   356→    version,
   357→   357→    latencyMs,
   358→   358→    ...(versionMatch
   359→   359→      ? {}
   360→   360→      : {
   361→   361→          reason: `"${a.modelName}" is not in the known model list for ${provider.label}.`,
   362→   362→          fixSteps: [
   363→   363→            `Verify the model name on the ${provider.label} dashboard.`,
   364→   364→            "Or pick a known model from the dropdown.",
   365→   365→          ],
   366→   366→        }),
   367→   367→  };
   368→   368→}

   370→   370→// ─── Limit / capacity check ────────────────────────────────────────────────
   371→   371→
   372→   372→export async function checkLimits(
   373→   373→  assignments: { id: string; label: string; assignment: ModelAssignment }[],
   374→   374→  estimatedTaskSize: "small" | "medium" | "large"
   375→   375→): Promise<ModelLimit[]> {
   376→   376→  const threshold =
   377→   377→    estimatedTaskSize === "large"
   378→   378→      ? 0.5
   379→   379→      : estimatedTaskSize === "medium"
   380→   380→      ? 0.25
   381→   381→      : 0.1;
   382→   382→
   383→   383→  return assignments.map(({ id, label, assignment }) => {
   384→   384→    if (assignment.connectionType === "API") {
   385→   385→      const remainingQuota = 0.7;
   386→   386→      const rateLimitPerMin = 60;
   387→   387→      const remainingTokens = 8000;
   388→   388→      const canFinish = remainingQuota >= threshold;
   389→   389→      return {
   390→   390→        id,
   391→   391→        label,
   392→   392→        connectionType: assignment.connectionType,
   393→   393→        provider: assignment.provider,
   394→   394→        modelName: assignment.modelName,
   395→   395→        remainingQuota,
   396→   396→        rateLimitPerMin,
   397→   397→        remainingTokens,
   398→   398→        canFinish,
   399→   399→        reason: canFinish
   400→   400→          ? undefined
   401→   401→          : `Quota ${Math.round(remainingQuota * 100)}% is below the ${Math.round(
   402→   402→              threshold * 100
   403→   403→            )}% threshold for a ${estimatedTaskSize} task.`,
   404→   404→      };
   405→   405→    } else {
   406→   406→      const busy = false;
   407→   407→      const estimatedCapacity = 0.85;
   408→   408→      const canFinish = !busy && estimatedCapacity >= threshold;
   409→   409→      return {
   410→   410→        id,
   411→   411→        label,
   412→   412→        connectionType: assignment.connectionType,
   413→   413→        provider: assignment.provider,
   414→   414→        modelName: assignment.modelName,
   415→   415→        busy,
   416→   416→        estimatedCapacity,
   417→   417→        canFinish,
   418→   418→        reason: canFinish
   419→   419→          ? undefined
   420→   420→          : busy
   421→   421→          ? "Local CLI is currently busy with another task."
   422→   422→          : `Estimated capacity ${Math.round(
   423→   423→              estimatedCapacity * 100
   424→   424→            )}% is below the ${Math.round(threshold * 100)}% threshold.`,
   425→   425→      };
   426→   426→    }
   427→   427→  });
   428→   428→}

   441→   441→function resolvePlan(
   442→   442→  doc: MultiModelConfigDoc,
   443→   443→  messages: ChatMessage[]
   444→   444→): {
   445→   445→  assignments: {
   446→   446→    id: string;
   447→   447→    label: string;
   448→   448→    assignment: ModelAssignment;
   449→   449→  }[];
   450→   450→  multiAgent: boolean;
   451→   451→  intent?: string;
   452→   452→} {
   453→   453→  const last = [...messages].reverse().find((m) => m.role === "user");
   454→   454→  const text = (last?.content || "").toLowerCase();
   455→   455→
   456→   456→  if (doc.mode === "SINGLE") {
   457→   457→    return {
   458→   458→      assignments: [
   459→   459→        {
   460→   460→          id: "global",
   461→   461→          label: "Single Model",
   462→   462→          assignment: doc.globalConfig || emptyAssignment(),
   463→   463→        },
   464→   464→      ],
   465→   465→      multiAgent: false,
   466→   466→    };
   467→   467→  }
   468→   468→
   469→   469→  if (doc.mode === "MULTI") {
   470→   470→    let feature: FeatureId = "chat";
   471→   471→    if (/(code|function|bug|class|api|sql|regex)/.test(text)) feature = "coding";
   472→   472→    else if (/(image|picture|photo|see|vision|ocr)/.test(text))
   473→   473→      feature = "vision";
   474→   474→    else if (/(voice|speak|speech|audio|transcri)/.test(text))
   475→   475→      feature = "voice";
   476→   476→    else if (/(automate|workflow|schedule|pipeline)/.test(text))
   477→   477→      feature = "automation";
   478→   478→    else if (/(robot|move|arm|sensor|actuator)/.test(text))
   479→   479→      feature = "robotics";
   480→   480→
   481→   481→    const assignment =
   482→   482→      doc.featureConfigs?.[feature] ||
   483→   483→      doc.featureConfigs?.chat ||
   484→   484→      emptyAssignment();
   485→   485→    return {
   486→   486→      assignments: [{ id: feature, label: feature, assignment }],
   487→   487→      multiAgent: false,
   488→   488→      intent: feature,
   489→   489→    };
   490→   490→  }
   491→   491→
   492→   492→  // ORCHESTRATOR
   493→   493→  const host = doc.hostConfig || emptyAssignment();
   494→   494→  let specialist: SpecialistId | null = null;
   495→   495→  if (/(plan|design|architect|roadmap|strategy|decompose)/.test(text))
   496→   496→    specialist = "planning";
   497→   497→  else if (/(code|function|bug|class|api|sql|regex|build|implement)/.test(text))
   498→   498→    specialist = "coding";
   499→   499→  else if (/(image|picture|photo|see|vision|ocr)/.test(text))
   500→   500→    specialist = "vision";
   501→   501→  else if (/(automate|workflow|schedule|pipeline|chain)/.test(text))
   502→   502→    specialist = "automation";
   503→   503→  else if (/(robot|move|arm|sensor|actuator)/.test(text))
   504→   504→    specialist = "robotics";
   505→   505→
   506→   506→  const assignments: {
   507→   507→    id: string;
   508→   508→    label: string;
   509→   509→    assignment: ModelAssignment;
   510→   510→  }[] = [{ id: "host", label: "Host", assignment: host }];
   511→   511→
   512→   512→  if (specialist) {
   513→   513→    const sp = doc.specialistConfigs?.[specialist] || emptyAssignment();
   514→   514→    assignments.push({ id: specialist, label: specialist, assignment: sp });
   515→   515→  }
   516→   516→
   517→   517→  return {
   518→   518→    assignments,
   519→   519→    multiAgent: !!specialist,
   520→   520→    intent: specialist || "general",
   521→   521→  };
   522→   522→}
   523→   523→
   524→   524→async function callModel(
   525→   525→  assignment: ModelAssignment,
   526→   526→  messages: ChatMessage[],
   527→   527→  opts: { timeoutMs: number; role: string; intent?: string }
   528→   528→): Promise<{
   529→   529→  output: string;
   530→   530→  latencyMs: number;
   531→   531→  retries: number;
   532→   532→  timedOut: boolean;
   533→   533→}> {
   534→   534→  const started = Date.now();
   535→   535→  let retries = 0;
   536→   536→  let timedOut = false;
   537→   537→
   538→   538→  for (let attempt = 0; attempt <= MAX_RETRY; attempt++) {
   539→   539→    try {
   540→   540→      const out = await Promise.race([
   541→   541→        realCall(assignment, messages, opts.role, opts.intent),
   542→   542→        new Promise<never>((_, reject) =>
   543→   543→          setTimeout(
   544→   544→            () => reject(new Error(`timeout after ${opts.timeoutMs}ms`)),
   545→   545→            opts.timeoutMs
   546→   546→          )
   547→   547→        ),
   548→   548→      ]);
   549→   549→      return {
   550→   550→        output: out,
   551→   551→        latencyMs: Date.now() - started,
   552→   552→        retries,
   553→   553→        timedOut: false,
   554→   554→      };
   555→   555→    } catch {
   556→   556→      retries = attempt + 1;
   557→   557→      timedOut = true;
   558→   558→      if (attempt === MAX_RETRY) break;
   559→   559→      await new Promise((r) => setTimeout(r, 200 * (attempt + 1)));
   560→   560→    }
   561→   561→  }
   562→   562→
   563→   563→  return {
   564→   564→    output: "",
   565→   565→    latencyMs: Date.now() - started,
   566→   566→    retries,
   567→   567→    timedOut,
   568→   568→  };
   569→   569→}

   571→   571→async function realCall(
   572→   572→  _assignment: ModelAssignment,
   573→   573→  messages: ChatMessage[],
   574→   574→  role: string,
   575→   575→  intent?: string
   576→   576→): Promise<string> {
   577→   577→  const ZAI = (await import("z-ai-web-dev-sdk")).default;
   578→   578→  const zai = await ZAI.create();
   579→   579→
   580→   580→  const systemHint =
   581→   581→    role === "host"
   582→   582→      ? "You are NOX Host. Analyze the user's intent and either answer directly or synthesize the response from a specialist model into a clean reply to the user. Be concise."
   583→   583→      : intent
   584→   584→      ? `You are NOX ${role} specialist (intent: ${intent}). Answer the user's request focused on your specialty. Be concise and useful.`
   585→   585→      : "You are NOX AI. Respond helpfully and concisely.";
   586→   586→
   587→   587→  const completion = await zai.chat.completions.create({
   588→   588→    messages: [
   589→   589→      { role: "assistant", content: systemHint },
   590→   590→      ...messages.map((m) => ({
   591→   591→        role:
   592→   592→          m.role === "system"
   593→   593→            ? ("assistant" as const)
   594→   594→            : (m.role as "user" | "assistant"),
   595→   595→        content: m.content,
   596→   596→      })),
   597→   597→    ],
   598→   598→    thinking: { type: "disabled" },
   599→   599→  });
   600→   600→
   601→   601→  return completion.choices[0]?.message?.content || "";
   602→   602→}

   604→   604→export async function dispatch(
   605→   605→  userId: string,
   606→   606→  messages: ChatMessage[],
   607→   607→  opts: { confirmMultiAgent?: boolean } = {}
   608→   608→): Promise<DispatchResult> {
   609→   609→  const doc = await getConfigInternal(userId);
   610→   610→  const plan = resolvePlan(doc, messages);
   611→   611→  const timeoutOverrides = doc.timeoutOverrides || {};
   612→   612→
   613→   613→  if (plan.multiAgent && !opts.confirmMultiAgent) {
   614→   614→    const limits = await checkLimits(plan.assignments, "medium");
   615→   615→    return {
   616→   616→      ok: false,
   617→   617→      mode: doc.mode,
   618→   618→      steps: [],
   619→   619→      finalReply: "",
   620→   620→      multiAgent: true,
   621→   621→      confirmationRequired: true,
   622→   622→      limits,
   623→   623→    };
   624→   624→  }
   625→   625→
   626→   626→  let limits: ModelLimit[] | undefined;
   627→   627→  if (plan.multiAgent && opts.confirmMultiAgent) {
   628→   628→    limits = await checkLimits(plan.assignments, "medium");
   629→   629→    const blocked = limits.find((l) => !l.canFinish);
   630→   630→    if (blocked) {
   631→   631→      return {
   632→   632→        ok: false,
   633→   633→        mode: doc.mode,
   634→   634→        steps: [],
   635→   635→        finalReply: "",
   636→   636→        multiAgent: true,
   637→   637→        confirmationRequired: false,
   638→   638→        limits,
   639→   639→        error: `Cannot run: "${blocked.label}" cannot complete its part. ${
   640→   640→          blocked.reason || ""
   641→   641→        }`,
   642→   642→      };
   643→   643→    }
   644→   644→  }
   645→   645→
   646→   646→  const steps: DispatchStep[] = [];
   647→   647→
   648→   648→  if (doc.mode === "ORCHESTRATOR" && plan.multiAgent) {
   649→   649→    const hostAssignment = plan.assignments[0].assignment;
   650→   650→    const hostTimeout =
   651→   651→      timeoutOverrides[hostAssignment.connectionType] ||
   652→   652→      DEFAULT_TIMEOUTS[hostAssignment.connectionType];
   653→   653→    const hostResult = await callModel(hostAssignment, messages, {
   654→   654→      timeoutMs: hostTimeout,
   655→   655→      role: "host",
   656→   656→      intent: plan.intent,
   657→   657→    });
   658→   658→
   659→   659→    const specialistAssignment = plan.assignments[1].assignment;
   660→   660→    const specialistTimeout =
   661→   661→      timeoutOverrides[specialistAssignment.connectionType] ||
   662→   662→      DEFAULT_TIMEOUTS[specialistAssignment.connectionType];
   663→   663→
   664→   664→    const specialistMessages: ChatMessage[] = [
   665→   665→      ...messages,
   666→   666→      {
   667→   667→        role: "assistant",
   668→   668→        content: `[Host routed this to the ${plan.assignments[1].label} specialist. Fulfill the request.]`,
   669→   669→      },
   670→   670→    ];
   671→   671→    const specialistResult = await callModel(
   672→   672→      specialistAssignment,
   673→   673→      specialistMessages,
   674→   674→      {
   675→   675→        timeoutMs: specialistTimeout,
   676→   676→        role: plan.assignments[1].label,
   677→   677→        intent: plan.intent,
   678→   678→      }
   679→   679→    );
   680→   680→
   681→   681→    const finalMessages: ChatMessage[] = [
   682→   682→      ...messages,
   683→   683→      {
   684→   684→        role: "assistant",
   685→   685→        content: `Specialist ${plan.assignments[1].label} responded with:\n\n${specialistResult.output}\n\nReply to the user, incorporating the specialist's work.`,
   686→   686→      },
   687→   687→    ];
   688→   688→    const finalResult = await callModel(hostAssignment, finalMessages, {
   689→   689→      timeoutMs: hostTimeout,
   690→   690→      role: "host",
   691→   691→    });
   692→   692→
   693→   693→    steps.push({
   694→   694→      role: "host",
   695→   695→      model: hostAssignment.modelName,
   696→   696→      provider: hostAssignment.provider,
   697→   697→      connectionType: hostAssignment.connectionType,
   698→   698→      intent: "analyze",
   699→   699→      input: messages[messages.length - 1]?.content || "",
   700→   700→      output: hostResult.output || "(host routing)",
   701→   701→      latencyMs: hostResult.latencyMs,
   702→   702→      retries: hostResult.retries,
   703→   703→      timedOut: hostResult.timedOut,
   704→   704→    });
   705→   705→    steps.push({
   706→   706→      role: plan.assignments[1].label,
   707→   707→      model: specialistAssignment.modelName,
   708→   708→      provider: specialistAssignment.provider,
   709→   709→      connectionType: specialistAssignment.connectionType,
   710→   710→      intent: plan.intent,
   711→   711→      input: "(routed by host)",
   712→   712→      output: specialistResult.output,
   713→   713→      latencyMs: specialistResult.latencyMs,
   714→   714→      retries: specialistResult.retries,
   715→   715→      timedOut: specialistResult.timedOut,
   716→   716→    });
   717→   717→    steps.push({
   718→   718→      role: "host",
   719→   719→      model: hostAssignment.modelName,
   720→   720→      provider: hostAssignment.provider,
   721→   721→      connectionType: hostAssignment.connectionType,
   722→   722→      intent: "synthesize",
   723→   723→      input: "(specialist response)",
   724→   724→      output: finalResult.output,
   725→   725→      latencyMs: finalResult.latencyMs,
   726→   726→      retries: finalResult.retries,
   727→   727→      timedOut: finalResult.timedOut,
   728→   728→    });
   729→   729→
   730→   730→    return {
   731→   731→      ok: true,
   732→   732→      mode: doc.mode,
   733→   733→      steps,
   734→   734→      finalReply: finalResult.output,
   735→   735→      multiAgent: true,
   736→   736→      confirmationRequired: false,
   737→   737→      limits,
   738→   738→    };
   739→   739→  }
   740→   740→
   741→   741→  // SINGLE or MULTI (single-model path)
   742→   742→  const a = plan.assignments[0].assignment;
   743→   743→  const timeout =
   744→   744→    timeoutOverrides[a.connectionType] || DEFAULT_TIMEOUTS[a.connectionType];
   745→   745→  const result = await callModel(a, messages, {
   746→   746→    timeoutMs: timeout,
   747→   747→    role: plan.assignments[0].label,
   748→   748→    intent: plan.intent,
   749→   749→  });
   750→   750→
   751→   751→  steps.push({
   752→   752→    role: plan.assignments[0].label,
   753→   753→    model: a.modelName,
   754→   754→    provider: a.provider,
   755→   755→    connectionType: a.connectionType,
   756→   756→    intent: plan.intent,
   757→   757→    input: messages[messages.length - 1]?.content || "",
   758→   758→    output: result.output,
   759→   759→    latencyMs: result.latencyMs,
   760→   760→    retries: result.retries,
   761→   761→    timedOut: result.timedOut,
   762→   762→  });
   763→   763→
   764→   764→  return {
   765→   765→    ok: true,
   766→   766→    mode: doc.mode,
   767→   767→    steps,
   768→   768→    finalReply: result.output,
   769→   769→    multiAgent: false,
   770→   770→    confirmationRequired: false,
   771→   771→  };
   772→   772→}
